cmake_minimum_required(VERSION 3.22.1)
project(tangowiz LANGUAGES CXX)
add_library(tangowiz SHARED jni_bridge.cpp recognition.cpp scanner.cpp mel_frontend.cpp)
target_compile_features(tangowiz PRIVATE cxx_std_17)
target_compile_options(tangowiz PRIVATE -O3 -fno-fast-math -ffp-contract=off)
if(ANDROID_ABI STREQUAL "x86_64")
 target_compile_options(tangowiz PRIVATE -mssse3)
endif()
target_link_libraries(tangowiz android log)
target_link_options(tangowiz PRIVATE -Wl,-z,max-page-size=16384)
